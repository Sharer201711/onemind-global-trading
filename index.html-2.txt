    async function loadCloudData() {
      try {
        const [sigRes, stratRes] = await Promise.all([
          fetch(`${FIREBASE_DB_URL}/signals.json`),
          fetch(`${FIREBASE_DB_URL}/strategies.json`)
        ]);

        if (!sigRes.ok) throw new Error('Failed to fetch signals');

        const sigData = await sigRes.json();
        if (sigData && typeof sigData === 'object') {
          signals = Object.entries(sigData)
            .map(([key, val]) => ({ ...val, id: key }))
            .sort((a, b) => (b.createdAt || 0) - (a.createdAt || 0));
        } else {
          // If database is brand new/empty, show sample signals until admin posts first one
          signals = [...defaultSignals];
        }

        const stratData = await stratRes.json();
        if (stratData && typeof stratData === 'object') {
          const customStrats = Object.entries(stratData).map(([key, val]) => ({ ...val, id: key }));
          strategies = [...customStrats, ...defaultStrategies];
        } else {
          strategies = [...defaultStrategies];
        }

        updateCloudBadge(true, "● Live Cloud Synced");
        renderSignals();
        renderStrategies();
      } catch (err) {
        console.error(err);
        updateCloudBadge(false, "● Cloud Error — Check Firebase Rules");
        renderSignals();
        renderStrategies();
      }
    }

    function formatMoney(num, symbol) {
      return symbol + num.toLocaleString('en-US', { minimumFractionDigits: 2, maximumFractionDigits: 2 });
    }

    function calculateCompounding() {
      const symbol = document.getElementById('calcCurrency').value;
      const interval = document.getElementById('calcInterval').value;
      const principal = Math.max(0, parseFloat(document.getElementById('calcPrincipal').value) || 0);
      const rate = parseFloat(document.getElementById('calcRate').value) || 0;
      const periods = Math.min(365, Math.max(1, parseInt(document.getElementById('calcPeriods').value) || 1));
      const deposit = Math.max(0, parseFloat(document.getElementById('calcDeposit').value) || 0);

      document.getElementById('principalLabel').textContent = formatMoney(principal, symbol);
      document.getElementById('rateLabel').textContent = rate + '%';
      document.getElementById('periodsLabel').textContent = periods;
      document.getElementById('periodUnitText').textContent = interval;
      document.getElementById('tableCountBadge').textContent = `${periods} ${interval}`;

      let balance = principal;
      let totalContributed = principal;
      const labels = ['Start'];
      const balances = [principal];
      const principalSeries = [principal];
      let rowsHtml = '';

      for (let i = 1; i <= periods; i++) {
        const starting = balance;
        const profit = starting * (rate / 100);
        balance = starting + profit + deposit;
        totalContributed += deposit;

        labels.push(`${interval.slice(0, 1)}${i}`);
        balances.push(Number(balance.toFixed(2)));
        principalSeries.push(Number(totalContributed.toFixed(2)));

        rowsHtml += `
          <tr class="hover:bg-white/5">
            <td class="py-2.5 px-4 text-gray-300 font-semibold">${interval.slice(0, -1)} ${i}</td>
            <td class="py-2.5 px-4 text-gray-400">${formatMoney(starting, symbol)}</td>
            <td class="py-2.5 px-4 text-emeraldPro-400">+${formatMoney(profit, symbol)}</td>
            <td class="py-2.5 px-4 text-right text-gold-400 font-bold">${formatMoney(balance, symbol)}</td>
          </tr>
        `;
      }

      const netProfit = balance - totalContributed;
      const growthPct = totalContributed > 0 ? ((netProfit / totalContributed) * 100) : 0;

      document.getElementById('resFinalBalance').textContent = formatMoney(balance, symbol);
      document.getElementById('resTotalProfit').textContent = '+' + formatMoney(netProfit, symbol);
      document.getElementById('resTotalGrowth').textContent = '+' + growthPct.toFixed(1) + '%';
      document.getElementById('compoundingTableBody').innerHTML = rowsHtml;

      updateChart(labels, balances, principalSeries, symbol);
    }

    function updateChart(labels, balances, principalSeries, symbol) {
      const ctx = document.getElementById('compoundingChart').getContext('2d');
      if (equityChart) equityChart.destroy();

      const gradient = ctx.createLinearGradient(0, 0, 0, 260);
      gradient.addColorStop(0, 'rgba(212, 175, 55, 0.35)');
      gradient.addColorStop(1, 'rgba(212, 175, 55, 0.0)');

      equityChart = new Chart(ctx, {
        type: 'line',
        data: {
          labels: labels,
          datasets: [
            {
              label: 'Compounded Balance',
              data: balances,
              borderColor: '#D4AF37',
              backgroundColor: gradient,
              borderWidth: 2.5,
              fill: true,
              tension: 0.35,
              pointRadius: labels.length > 25 ? 0 : 3,
              pointBackgroundColor: '#F8E08E'
            },
            {
              label: 'Principal + Deposits',
              data: principalSeries,
              borderColor: '#10B981',
              borderDash: [5, 5],
              borderWidth: 1.5,
              fill: false,
              tension: 0.1,
              pointRadius: 0
            }
          ]
        },
        options: {
          responsive: true,
          maintainAspectRatio: false,
          interaction: { mode: 'index', intersect: false },
          plugins: {
            legend: {
              labels: { color: '#9CA3AF', font: { family: 'Inter', size: 11 } }
            },
            tooltip: {
              backgroundColor: '#0D131F',
              borderColor: 'rgba(212, 175, 55, 0.4)',
              borderWidth: 1,
              callbacks: {
                label: (ctx) => `${ctx.dataset.label}: ${symbol}${ctx.parsed.y.toLocaleString()}`
              }
            }
          },
          scales: {
            x: {
              grid: { color: 'rgba(255,255,255,0.04)' },
              ticks: { color: '#6B7280', maxTicksLimit: 10, font: { family: 'JetBrains Mono', size: 10 } }
            },

              }
            }
          }
        }
      });
    }

    function downloadLogoSVG() {
      const svgEl = document.getElementById('brandLogoSvg');
      const serializer = new XMLSerializer();
      const source = serializer.serializeToString(svgEl);
      const blob = new Blob([source], { type: 'image/svg+xml;charset=utf-8' });
      const url = URL.createObjectURL(blob);
      const link = document.createElement('a');
      link.href = url;
      link.download = 'Onemind-Global-Trading-Logo.svg';
      document.body.appendChild(link);
      link.click();
      document.body.removeChild(link);
      URL.revokeObjectURL(url);
      showToast('Logo SVG downloaded!');
    }
